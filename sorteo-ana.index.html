<script>

    // Función para lanzar el confeti

    function dispararConfeti() {

        var duration = 3 * 1000; // El confeti durará 3 segundos

        var end = Date.now() + duration;

 

        (function frame() {

            // Colores de la marca de Instagram

            var coloresInstagram = ['#405DE6', '#5851DB', '#833AB4', '#C13584', '#E1306C', '#FD1D1D', '#F56040', '#F77737', '#FCAF45', '#FFDC80'];

 

            confetti({

                particleCount: 5,

                angle: 60,

                spread: 55,

                origin: { x: 0 },

                colors: coloresInstagram

            });

            confetti({

                particleCount: 5,

                angle: 120,

                spread: 55,

                origin: { x: 1 },

                colors: coloresInstagram

            });

 

            if (Date.now() < end) {

                requestAnimationFrame(frame);

            }

        }());

    }

 

    function startDraw() {

        const textArea = document.getElementById('participants').value;

        const participants = textArea.split('\n').filter(name => name.trim() !== '');

 

        if (participants.length < 3) {

            alert("Necesitas al menos 3 participantes para el sorteo.");

            return;

        }

 

        document.getElementById('setup-section').style.display = 'none';

        document.getElementById('loader').style.display = 'block';

        document.getElementById('loading-text').style.display = 'block';

 

        // TIEMPO DE ESPERA (15000 = 15 segundos)

        const tiempoDeSpeech = 15000;

 

        setTimeout(() => {

            const shuffled = participants.sort(() => 0.5 - Math.random());

            const winners = shuffled.slice(0, 3);

 

            document.getElementById('loader').style.display = 'none';

            document.getElementById('loading-text').style.display = 'none';

 

            const winnersListDiv = document.getElementById('winners-list');

            winnersListDiv.innerHTML = '';

           

            winners.forEach((winner, index) => {

                const div = document.createElement('div');

                div.className = 'winner-name';

                div.innerText = winner;

                // Pequeño retraso entre cada nombre para que no aparezcan de golpe

                div.style.animationDelay = `${index * 0.3}s`;

                winnersListDiv.appendChild(div);

            });

 

            document.getElementById('winners-section').style.display = 'block';

           

            // ¡Lanzar el confeti!

            dispararConfeti();

 

        }, tiempoDeSpeech);

    }

</script>
